            // Real-time Exam Countdown Timer
            function updateCountdown() {
                const now = new Date();
                const examDate = new Date(customExamDateStr);
                const diff = examDate - now;
                if (diff > 0) {
                    const days = Math.floor(diff / (1000 * 60 * 60 * 24));
                    const hours = Math.floor((diff / (1000 * 60 * 60)) % 24);
                    const mins = Math.floor((diff / (1000 * 60)) % 60);
                    
                    const dEl = document.getElementById('countdown-days');
                    const hEl = document.getElementById('countdown-hours');
                    const mEl = document.getElementById('countdown-mins');
                    if (dEl) dEl.innerText = days;
                    if (hEl) hEl.innerText = hours;
                    if (mEl) mEl.innerText = mins;
                }
            }
            updateCountdown();
            setInterval(updateCountdown, 1000);
        };

                if (loginBox) loginBox.classList.add('hidden');
                if (gridBox) gridBox.classList.remove('hidden');
                if (statusBar) {
                    statusBar.innerHTML = `
                        <div class="px-4 py-2.5 rounded-2xl bg-indigo-500/20 text-indigo-700 dark:text-indigo-300 text-xs font-bold flex justify-between items-center border border-indigo-500/30">
                            <span>${role === 'pubudu' ? 'Owner Admin' : 'Sub-Admin'}</span>
                            <button onclick="logoutPageAdmin()" class="text-rose-500 hover:underline text-[11px] font-bold ml-3">ඉවත් වන්න</button>
                        </div>
                    `;
                }
                if (role === 'assistant' && lockOverlay) {
                    lockOverlay.classList.remove('hidden');
                } else if (lockOverlay) {
                    lockOverlay.classList.add('hidden');
                }
            } else {
                if (loginBox) loginBox.classList.remove('hidden');
                if (gridBox) gridBox.classList.add('hidden');
                if (statusBar) statusBar.innerHTML = '';
            }
        }

        function submitPageNoteForm() {
            const sub = document.getElementById('form-note-subject').value;
            const grade = document.getElementById('form-note-grade').value;
            const title = document.getElementById('form-note-title').value.trim();
            const shortDesc = document.getElementById('form-note-shortdesc').value.trim();
            const pdfLink = document.getElementById('form-note-pdflink').value.trim();
            const content = document.getElementById('form-note-content').value.trim();

            if (!title) {
                showToast('කරුණාකර පාඩමේ මාතෘකාව ඇතුළත් කරන්න.', 'අවවාදයයි');
                return;
            }

            let lessons = JSON.parse(localStorage.getItem('al_commerce_lessons')) || [];
            lessons.unshift({
                id: Date.now(),
                subject: sub,
                grade: parseInt(grade),
                title,
                shortDesc: shortDesc || title,
                content: content || 'පාඩම් සටහන අඩංගු වේ.',
                pdfLink: pdfLink || '#',
                isFree: false
            });
            localStorage.setItem('al_commerce_lessons', JSON.stringify(lessons));

            document.getElementById('form-note-title').value = '';
            document.getElementById('form-note-shortdesc').value = '';
            document.getElementById('form-note-pdflink').value = '';
            document.getElementById('form-note-content').value = '';

            renderLessons();
            showToast('පාඩම් සටහන සාර්ථකව ප්‍රකාශයට පත් කරන ලදී!', 'සාර්ථකයි');
        }

        function submitPageFlashcardForm() {
            const subject = document.getElementById('form-flash-subject').value;
            const front = document.getElementById('form-flash-front').value.trim();
            const back = document.getElementById('form-flash-back').value.trim();

            if (!front || !back) {
                showToast('කරුණාකර ප්‍රශ්නය සහ පිළිතුර දෙකම ඇතුළත් කරන්න.', 'අවවාදයයි');
                return;
            }

            let subKey = 'bs';
            if (subject.includes('ගිණුම්කරණය')) subKey = 'acc';
            else if (subject.includes('ආර්ථික')) subKey = 'econ';
            else if (subject.includes('තොරතුරු')) subKey = 'ict';
            else if (subject.includes('පොදු')) subKey = 'git';

            flashcardsData.unshift({ subject, key: subKey, front, back });
            localStorage.setItem('al_commerce_flashcards', JSON.stringify(flashcardsData));

            document.getElementById('form-flash-front').value = '';
            document.getElementById('form-flash-back').value = '';

            renderFlashcard();
            renderSubjectFlashcard();
            showToast('නව ෆ්ලෑෂ් කාඩ්පත සාර්ථකව එකතු කරන ලදී!', 'සාර්ථකයි');
        }
